cmake_minimum_required(VERSION 3.16)

project(comp3722_parser
        VERSION 1.0
        DESCRIPTION "SLR parser for the extended C-subset OO language"
        LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(CheckCXXCompilerFlag)

add_executable(parser
        Grammar.cpp
        SLRTable.cpp
        Token.cpp
        ParseTree.cpp
        Parser.cpp
        main.cpp)

target_include_directories(parser PRIVATE ..)

add_executable(lexer
        ../lexer.c)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(parser PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(lexer PRIVATE -Wall -Wextra -Wpedantic)
endif()
